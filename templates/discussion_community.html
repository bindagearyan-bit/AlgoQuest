For Python questers: remember arr[l], arr[r] = arr[r], arr[l] is done via tuple packing under the hood, but in C++ you want explicit std::swap to maintain strict O(1) auxiliary overhead.

void reverseArray(vector<int>& arr) {
  int l = 0, r = arr.size() - 1;
  while (l < r) { std::swap(arr[l++], arr[r--]); }
}
